// Bluecat BLE-Sensor für Shelly

// Der Override hält die ID mit config/shelly_wohnzimmer.json synchron.

// Leer lassen => ID und Topic werden aus den Shelly-Geräteinformationen erzeugt.

let sensor_id_override = "shelly_wohnzimmer_dim_sb";

let sensor_name_override = "Shelly WZ Dim SB";

// Alte Discovery-ID aus der vorherigen Konfiguration. Beim ersten

// MQTT-Verbindungsaufbau wird deren retained Discovery-Konfiguration gelöscht.

let legacy_sensor_id = "shelly_wohnzimmer_dim_sb";

let target_mac = "c6:5c:bd:1f:77:56";

// MACs der Sensoren, deren Standard-Advertisements als Mesh gelten.

let mesh_peer_macs = ["8c:bf:ea:97:98:4a"];

let min_publish_interval_ms = 2000;

let timeout_ms = 30000;

let offline_rssi = -130;
let mesh_advertising_enabled = true;
let mesh_advertising_interval_ms = 8000; // --- A: Auf 8 Sekunden erhöht ---

// --- Interne Variablen für die neue Architektur ---
let sequence = 0;
let mesh_sequences = {};
let last_mesh_publish_ms = {};
let mesh_prefix = "TRILOLA_SENSOR:";

// Der O(1) Lookup-Table (Wörterbuch) für sofortige MAC-Zuordnung
let fast_mac_lookup = {};

// Event-Timer Variablen
let offline_timer = null;
let batch_timer = null;
let lola_rssi_batch = []; // Sammelt alle Pings für die Median-Berechnung

print("Starte hocheffizientes TriLola Skript...");

// --- Hilfsfunktionen ---
function sanitize(value) {
  let str = String(value || "shelly_sensor").toLowerCase();
  let result = "";
  for (let i = 0; i < str.length; i++) {
    let c = str[i];
    if ((c >= "a" && c <= "z") || (c >= "0" && c <= "9") || c === "_" || c === "-") {
      result += c;
    } else {
      result += "_";
    }
  }
  return result;
}

function normalize_mac(value) {
  let str = String(value || "").toLowerCase();
  let result = "";
  for (let i = 0; i < str.length; i++) {
    let c = str[i];
    // Formatiert strikt zu aa:bb:cc:dd:ee:ff für den Lookup
    if (c !== ":" && c !== "-") {
      result += c;
      if (result.length % 3 === 2 && result.length < 17) result += ":";
    }
  }
  return result;
}

// Baut das Wörterbuch (Lookup-Table) auf, um Schleifen beim Scannen zu vermeiden
function rebuild_mac_lookup() {
  let new_lookup = {};
  if (target_mac) {
    new_lookup[normalize_mac(target_mac)] = "target";
  }
  for (let i = 0; i < mesh_peer_macs.length; i++) {
    if (mesh_peer_macs[i]) {
      new_lookup[normalize_mac(mesh_peer_macs[i])] = "mesh";
    }
  }
  fast_mac_lookup = new_lookup;
  print(sensor_name + ": Filterliste aktualisiert.");
}

// Shelly-Geräteinfos laden
let device_info = Shelly.getDeviceInfo();
let generated_id = sanitize(device_info.id || device_info.mac || device_info.name);
let sensor_id = sanitize(sensor_id_override || generated_id);
let sensor_name = sensor_name_override || device_info.name || ("Shelly " + sensor_id);
let sensor_ble_mac = String(device_info.mac || "").toLowerCase();

// MQTT Topics
let identity_topic = "bluecat/registry/" + sensor_id + "/identity";
let mesh_peers_topic = "bluecat/registry/mesh_peers";
let target_mac_state_topic = "bluecat/config/target_mac/state";
let mqtt_topic = "bluecat/" + sensor_id + "/sensor/state";
let availability_topic = "bluecat/" + sensor_id + "/sensor/status";
let discovery_sensor = "homeassistant/sensor/bluecat_" + sensor_id + "_rssi/config";
let discovery_presence = "homeassistant/binary_sensor/bluecat_" + sensor_id + "_presence/config";

// --- MQTT & Config ---
function publish_identity() {
  if (!sensor_ble_mac) return;
  MQTT.publish(identity_topic, JSON.stringify({
    schema: 1, sensor_id: sensor_id, name: sensor_name,
    implementation: "shelly", state_topic: mqtt_topic,
    availability_topic: availability_topic, ble_mac: sensor_ble_mac,
    enabled: true, timestamp: Math.floor(Date.now() / 1000)
  }), 1, true);
}

function update_runtime_config(topic, message) {
  let changed = false;
  if (topic === target_mac_state_topic) {
    try {
      let value = JSON.parse(message);
      target_mac = String(value.target_mac || value.mac || "");
    } catch (error) {
      target_mac = String(message || "");
    }
    changed = true;
  }
  if (topic === mesh_peers_topic) {
    try {
      let data = JSON.parse(message);
      if (data && Array.isArray(data.peers)) {
        let peers = [];
        data.peers.forEach(function (peer) {
          if (peer && peer.ble_mac) peers.push(peer.ble_mac);
        });
        mesh_peer_macs = peers;
        changed = true;
      }
    } catch (error) { }
  }
  if (changed) rebuild_mac_lookup();
}

// --- MEDIAN BATCHING LOGIK (Punkt D) ---
function get_median(arr) {
  if (arr.length === 0) return offline_rssi;
  // Simpler, ressourcenschonender Bubble-Sort für kleine Arrays (perfekt für mJS)
  let len = arr.length;
  for (let i = 0; i < len; i++) {
    for (let j = 0; j < len - i - 1; j++) {
      if (arr[j] > arr[j + 1]) {
        let temp = arr[j];
        arr[j] = arr[j + 1];
        arr[j + 1] = temp;
      }
    }
  }
  let mid = Math.floor(len / 2);
  return (len % 2 === 0) ? (arr[mid - 1] + arr[mid]) / 2.0 : arr[mid];
}

function flush_lola_batch() {
  if (lola_rssi_batch.length === 0) {
    batch_timer = null; // Timer stoppen, wenn nichts empfangen wurde
    return;
  }
  
  let median_rssi = get_median(lola_rssi_batch);
  let sample_count = lola_rssi_batch.length;
  lola_rssi_batch = []; // Batch leeren für die nächste Runde
  sequence += 1;

  MQTT.publish(mqtt_topic, JSON.stringify({
    message_type: "tag_rssi", sensor_id: sensor_id,
    rssi: median_rssi, timestamp: Math.floor(Date.now() / 1000),
    sequence: sequence, sample_count: sample_count, present: true
  }), 0, false);
  
  // Timer neu aufziehen für die nächste Batch-Runde
  batch_timer = Timer.set(min_publish_interval_ms, false, flush_lola_batch);
}

// --- EVENT-TIMER LOGIK (Punkt C) ---
function mark_target_offline() {
  offline_timer = null; // Timer ist abgelaufen
  sequence += 1;
  MQTT.publish(mqtt_topic, JSON.stringify({
    message_type: "tag_rssi", sensor_id: sensor_id,
    rssi: offline_rssi, timestamp: Math.floor(Date.now() / 1000),
    sequence: sequence, sample_count: 1, present: false
  }), 0, false);
  print(sensor_name + ": Ziel offline (Timeout)");
}

function reset_offline_timer() {
  if (offline_timer !== null) {
    Timer.clear(offline_timer); // Vorherigen Timer abwürgen
  }
  // Neuen 30-Sekunden Timer starten
  offline_timer = Timer.set(timeout_ms, false, mark_target_offline);
}


// --- MESH LOGIK ---
function decode_mesh_sensor_id(res) {
  let candidates = [];
  if (res.name) candidates.push(String(res.name));
  if (res.local_name) candidates.push(String(res.local_name));
  let manufacturer_data = res.manufacturer_data || res.manufacturerData;
  if (manufacturer_data) {
    Object.keys(manufacturer_data).forEach(function (key) {
      let value = manufacturer_data[key];
      if (typeof value === "string") candidates.push(value);
      else if (Array.isArray(value)) {
        let text = "";
        value.forEach(function (byte) { text += String.fromCharCode(byte); });
        candidates.push(text);
      }
    });
  }
  for (let i = 0; i < candidates.length; i++) {
    if (candidates[i].indexOf(mesh_prefix) === 0) {
      return candidates[i].slice(mesh_prefix.length).trim();
    }
  }
  return "";
}

function publish_mesh_payload(beacon_mac, rssi, legacy_sensor_id) {
  let now = Date.now();
  if (now - (last_mesh_publish_ms[beacon_mac] || 0) < min_publish_interval_ms) return;
  mesh_sequences[beacon_mac] = (mesh_sequences[beacon_mac] || 0) + 1;
  
  let payload = {
    message_type: "sensor_beacon", sensor_id: sensor_id,
    beacon_mac: beacon_mac, rssi: rssi,
    timestamp: Math.floor(Date.now() / 1000),
    sequence: mesh_sequences[beacon_mac]
  };
  if (legacy_sensor_id && legacy_sensor_id !== sensor_id) {
    payload.beacon_sensor = legacy_sensor_id;
  }
  MQTT.publish(mqtt_topic, JSON.stringify(payload), 0, false);
  last_mesh_publish_ms[beacon_mac] = now;
}

function advertise_mesh_once() {
  if (!mesh_advertising_enabled) return;
  Shelly.call("BLE.AdvertiseOnce", { adv_data: "020106" }, function () {});
}

// --- MQTT SETUP ---
function publish_discovery() {
  let device = { identifiers: ["bluecat_sensor_" + sensor_id], name: sensor_name, manufacturer: "Bluecat", model: "Shelly BLE Sensor" };
  MQTT.publish(discovery_sensor, JSON.stringify({
    name: sensor_name + " RSSI", unique_id: "bluecat_" + sensor_id + "_rssi", state_topic: mqtt_topic, availability_topic: availability_topic,
    value_template: "{% if value_json.message_type is not defined or value_json.message_type == 'tag_rssi' %}{{ value_json.rssi }}{% endif %}",
    unit_of_measurement: "dBm", device_class: "signal_strength", entity_category: "diagnostic", device: device
  }), 1, true);
  MQTT.publish(discovery_presence, JSON.stringify({
    name: sensor_name + " Präsenz", unique_id: "bluecat_" + sensor_id + "_presence", state_topic: mqtt_topic, availability_topic: availability_topic,
    value_template: "{% if value_json.message_type is not defined or value_json.message_type == 'tag_rssi' %}{{ 'ON' if value_json.present is defined and value_json.present else ('ON' if value_json.rssi|float > -120 else 'OFF') }}{% endif %}",
    payload_on: "ON", payload_off: "OFF", device_class: "presence", entity_category: "diagnostic", device: device
  }), 1, true);
}

function on_mqtt_connect() {
  print(sensor_name + ": MQTT verbunden!");
  if (legacy_sensor_id && legacy_sensor_id !== sensor_id) {
    MQTT.publish("homeassistant/sensor/bluecat_" + legacy_sensor_id + "_rssi/config", "", 1, true);
    MQTT.publish("homeassistant/binary_sensor/bluecat_" + legacy_sensor_id + "_presence/config", "", 1, true);
  }
  MQTT.publish(availability_topic, "online", 1, true);
  MQTT.subscribe(mesh_peers_topic, update_runtime_config);
  MQTT.subscribe(target_mac_state_topic, update_runtime_config);
  publish_identity();
  publish_discovery();
}

if (MQTT.isConnected()) on_mqtt_connect();
MQTT.setConnectHandler(on_mqtt_connect);

// Lookup initialisieren
rebuild_mac_lookup();

// Start Advertising Timer
Timer.set(60000, true, publish_identity);
Timer.set(mesh_advertising_interval_ms, true, advertise_mesh_once);
advertise_mesh_once();

// --- DER NEUE, ULTRASCHNELLE SCANNER (Punkte 1 & 3) ---
BLE.Scanner.Start({ duration_ms: BLE.Scanner.INFINITE_SCAN, active: false });

BLE.Scanner.Subscribe(function (ev, res) {
  if (ev !== BLE.Scanner.SCAN_RESULT) return;

  // 1. MAC formatieren und O(1) Lookup durchführen
  let mac = String(res.addr).toLowerCase();
  let role = fast_mac_lookup[mac];

  // 2. FAST FAIL: Ist das Gerät relevant?
  if (!role) return; // Stiller Tod für alle fremden Geräte in Mikrosekunden

  // 3. Verarbeitung Lola (Ziel)
  if (role === "target") {
    lola_rssi_batch.push(res.rssi); // Ping in den Batch legen
    reset_offline_timer();          // Event-Timer zurücksetzen
    
    // Batch-Versand starten, falls er schläft
    if (batch_timer === null) {
      batch_timer = Timer.set(min_publish_interval_ms, false, flush_lola_batch);
    }
  } 
  // 4. Verarbeitung Mesh
  else if (role === "mesh") {
    publish_mesh_payload(mac, res.rssi, decode_mesh_sensor_id(res));
  }
});
